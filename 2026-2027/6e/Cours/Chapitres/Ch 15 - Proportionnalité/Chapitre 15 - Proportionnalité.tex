\subsection*{I. Situation de proportionnalité}
\addcontentsline{toc}{subsection}{I. Situation de proportionnalité}

\begin{Définition}[c]
Deux grandeurs sont \textbf{proportionnelles} lorsque l'on peut passer des valeurs de l'une aux valeurs de l'autre en multipliant par \textbf{un même nombre non nul}, appelé \textbf{coefficient de proportionnalité}. Pour les représenter, on utilise un \textbf{tableau de proportionnalité}.
\end{Définition}

\begin{Exemples}
\begin{enumerate}
\item À 10 ans, Jade mesurait 1,20 m. Aujourd'hui, elle a 20 ans et elle mesure 1,70 m.\\
La taille de Jade est-elle proportionnelle à son âge ? \\
Comme $2 \times 10\text{ ans}= 20\text{ ans}$ mais $2 \times 1,20\text{ m}= 2,40\text{ m} \neq 1,70\text{ m}$. \\
La taille de Jade n'est pas proportionnelle à son âge.
\item Au marché, 2 kg de bananes coûtent 5 € et 6 kg coûtent 15 €.\\
Le prix des bananes est-il proportionnel à la quantité achetée ?\\
Comme $3 \times 2 \text{ kg} = 6 \text{ kg}$ et que $3 \times 5 \text{ €}= 15 \text{ €}$\\
Le prix des bananes est proportionnel à la quantité achetée.
\end{enumerate}
\end{Exemples}

\begin{Exemple}
Le tableau donne le prix par kilo de pommes achetées. \\
Le prix est-il proportionnel à la masse de pommes achetées ?
\begin{center}
\begin{tikzpicture}
    % Tableau principal  
    \node at (0,0) { \arrayrulecolor{Couleur8}
        \begin{tabular}{|>{\centering\arraybackslash}p{3cm}|>{\centering\arraybackslash}p{1cm}|>{\centering\arraybackslash}p{1cm}|>{\centering\arraybackslash}p{1cm}|>{\centering\arraybackslash}p{1cm}|}
            \hline
            \cellcolor{Couleur8!10} Masse (en kg) & 1 & 2 & 6 \\
            \hline
            \cellcolor{Couleur8!10}Prix (en €) & 2,50 & 5 & 15  \\
            \hline
        \end{tabular}
    };
    
    % Flèches et annotations
    \draw[->, thick] (4.5, 0.3) to[bend left=45] (4.5, -0.3);
    \node at (5.5, 0) {$\times 2,5$};
    
    \draw[->, thick] (-4.5, -0.3) to[bend left=45] (-4.5, 0.3);
    \node at (-5.5, 0) {$\div 2,5$};
    
\end{tikzpicture}
\end{center}
$\dfrac{2,50}{1}=2,50$ ; $\dfrac{5}{2}=2,50$ et $\dfrac{15}{6}=2,50$.\\

TOUS les rapports étant égaux, le prix est bien proportionnel à la masse de pommes achetées. On dit que 2,5 est le \textbf{coefficient de proportionnalité}. 
\end{Exemple}

\begin{Définition}[c]
La \textbf{proportion} d'une partie d'un ensemble est définie par le quotient :
$$\text{Proportion} = \dfrac{\text{Nombre d'éléments de la partie}}{\text{Nombre total d'éléments}}$$
\end{Définition}

\begin{Exemple}
Dans un sac, il y a deux boules blanches et huit boules noires. \\
Quelle est la proportion de boules blanches ?\\
La proportion de boules blanches dans le sac est donc $\dfrac{2}{10} = 0,2 = 20\,\%$.
\end{Exemple}

\begin{Remarque}
Une proportion est toujours inférieure ou égale à 1.
\end{Remarque}

\subsection*{III. Utiliser une échelle}
\addcontentsline{toc}{subsection}{III. Utiliser une échelle}

\begin{Définition}
L'\textbf{échelle} d'un plan est le coefficient de proportionnalité entre les distances représentées sur un plan et les distances réelles, \textbf{exprimées dans la même unité}.
$$\text{Échelle} = \dfrac{\text{Distance sur le plan}}{\text{Distance réelle}}$$
\end{Définition}

\begin{Exemple}
Sur un plan à l'échelle $\dfrac{1}{100\,000}$, une longueur de 3,4 cm représente quelle distance dans la vraie vie ?\\
\begin{center}
\begin{tikzpicture}
\arrayrulecolor{Couleur10}
    % Tableau principal avec fond bleu clair
    \node[draw=none, inner sep=0pt, outer sep=0pt, anchor=center] at (0,0) {
        \begin{tabular}{|>{\centering\arraybackslash}m{7cm}|>{\centering\arraybackslash}m{1.5cm}|>{\centering\arraybackslash}m{1.5cm}|}
            \hline
            \cellcolor{Couleur8!10}Distance sur la carte (en cm) & 1 & 3,4 \\
            \hline
            \cellcolor{Couleur8!10}Distance dans la réalité (en cm) & 100\,000 & 340\,000 \\
            \hline
        \end{tabular}
    };
    
    % Flèches et annotations
    \draw[->, thick] (5.8, 0.3) to[bend left=45] (5.8, -0.3);
    \node at (7, 0) {$\times 100\,000$};
    
    \draw[->, thick] (-5.8, -0.3) to[bend left=45] (-5.8, 0.3);
    \node at (-7, 0) {$\div 100\,000$};
    
\end{tikzpicture}
\end{center}
3,4 cm carte correspondent alors à 340 000 cm dans la réalité, soit 3,4 km.

\end{Exemple}
